set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SPICE_BUILD_UI "Build the native SDL/OpenGL application" ON)
option(SPICE_WEBOS "Build for webOS; requires a cross toolchain and real SDK headers" OFF)
option(SPICE_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(SPICE_STRICT "Treat compiler warnings as errors" ON)
find_package(Threads REQUIRED)
find_package(PNG REQUIRED)
find_package(ZLIB REQUIRED)
find_path(JSONC_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSONC_LIBRARY NAMES json-c libjson-c REQUIRED)
add_library(spice_core src/core.cpp src/services.cpp src/designs.cpp src/prepared.cpp src/inventory.cpp src/theme.cpp src/persistence.cpp)
target_include_directories(spice_core PUBLIC include ${JSONC_INCLUDE_DIR})
target_link_libraries(spice_core PUBLIC ${JSONC_LIBRARY} Threads::Threads PNG::PNG ZLIB::ZLIB)
find_path(SPICE_CURL_INCLUDE_DIR curl/curl.h)
if(NOT CMAKE_CROSSCOMPILING AND UNIX AND NOT APPLE)
  find_library(SPICE_CURL_LIBRARY NAMES curl libcurl PATHS /usr/lib/${CMAKE_LIBRARY_ARCHITECTURE} NO_DEFAULT_PATH)
endif()
if(NOT SPICE_CURL_LIBRARY)
  find_package(CURL QUIET)
endif()
if(SPICE_CURL_LIBRARY AND SPICE_CURL_INCLUDE_DIR)

  target_link_libraries(spice_core PRIVATE ${SPICE_CURL_LIBRARY})
  target_compile_definitions(spice_core PRIVATE SPICE_HAVE_CURL=1)

  target_link_libraries(spice_core PRIVATE CURL::libcurl)
  target_compile_definitions(spice_core PRIVATE SPICE_HAVE_CURL=1)

  message(WARNING "libcurl absent: RetroPie HTTP disabled, HDMI handoff remains available")

if(SPICE_WEBOS)
  if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^[aA][rR][mM]")
    message(FATAL_ERROR "SPICE_WEBOS requires a real ARM cross-toolchain, not a host executable")
  endif()
  target_compile_definitions(spice_core PUBLIC SPICE_WEBOS=1)
endif()
function(spice_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    if(SPICE_STRICT)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wformat=2)
    if(SPICE_STRICT)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
    if(SPICE_SANITIZE)
      target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
  endif()
endfunction()
spice_warnings(spice_core)

  target_link_libraries(spice_inventory_tests PRIVATE spice_core)

if(SPICE_BUILD_UI)
  find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2)
  find_path(FREETYPE_INCLUDE_DIR ft2build.h PATH_SUFFIXES freetype2)
  if(SPICE_WEBOS AND (NOT SDL2_INCLUDE_DIR OR NOT FREETYPE_INCLUDE_DIR))
    message(FATAL_ERROR "Target SDL2 and FreeType SDK headers are required for webOS")
  endif()
  add_executable(spice-home-cpp src/main.cpp src/render.cpp src/ui.cpp src/ui_designs.cpp src/ui_library.cpp src/ui_themes.cpp src/ui_persistence.cpp src/ui_scenes.cpp src/runtime.cpp)
  target_compile_definitions(spice-home-cpp PRIVATE SPICE_BUILD_CONFIGURATION="$<CONFIG>" SPICE_SANITIZER_BUILD=$<BOOL:${SPICE_SANITIZE}>)
  target_include_directories(spice-home-cpp PRIVATE compat)
  if(SDL2_INCLUDE_DIR AND FREETYPE_INCLUDE_DIR)
    target_include_directories(spice-home-cpp PRIVATE ${SDL2_INCLUDE_DIR} ${FREETYPE_INCLUDE_DIR})
    target_compile_definitions(spice-home-cpp PRIVATE SPICE_SYSTEM_HEADERS=1)
  else()
    message(WARNING "Using narrow desktop-only SDL2/FreeType ABI declarations. Runtime checks enabled; webOS refuses this mode.")
  endif()
  target_link_libraries(spice-home-cpp PRIVATE spice_core PNG::PNG ${CMAKE_DL_LIBS})
  set_target_properties(spice-home-cpp PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
  spice_warnings(spice-home-cpp)
  # Native framebuffer checks need a real GL context. Keep cross/TV builds and
  # hosts without Xvfb outside these desktop tests. Their leak mode is explicit:
  # the separately tracked SDL/libdbus startup allocation is not a clean LSAN pass.
  if(BUILD_TESTING AND UNIX AND NOT CMAKE_CROSSCOMPILING AND NOT SPICE_WEBOS)
    find_program(SPICE_XVFB_RUN xvfb-run)
    foreach(check IN ITEMS performance_renderer_agent batch_renderer_checks backdrop_renderer_checks theme_renderer_checks input_order_checks home_cache_renderer_checks)
      add_executable(spice_${check} tests/${check}.cpp src/render.cpp src/runtime.cpp)
      if(check STREQUAL "input_order_checks")
        target_sources(spice_${check} PRIVATE src/ui.cpp src/ui_designs.cpp src/ui_library.cpp src/ui_themes.cpp src/ui_persistence.cpp src/ui_scenes.cpp)
      endif()
      target_include_directories(spice_${check} PRIVATE compat)
      if(SDL2_INCLUDE_DIR AND FREETYPE_INCLUDE_DIR)
        target_include_directories(spice_${check} PRIVATE ${SDL2_INCLUDE_DIR} ${FREETYPE_INCLUDE_DIR})
        target_compile_definitions(spice_${check} PRIVATE SPICE_SYSTEM_HEADERS=1)
      endif()
      target_link_libraries(spice_${check} PRIVATE spice_core PNG::PNG ${CMAKE_DL_LIBS})
      spice_warnings(spice_${check})
      if(SPICE_XVFB_RUN)
        set(check_arguments)
        if(check STREQUAL "batch_renderer_checks")
          set(check_arguments "${CMAKE_BINARY_DIR}/batch-comparison")
        elseif(check STREQUAL "backdrop_renderer_checks")
          set(check_arguments "${CMAKE_BINARY_DIR}/backdrop-comparison")
        elseif(check STREQUAL "theme_renderer_checks")
          set(check_arguments "${CMAKE_SOURCE_DIR}/demos/themes/demo-data/themes" "${CMAKE_BINARY_DIR}/theme-renderer-check")
        elseif(check STREQUAL "input_order_checks")
          set(check_arguments "${CMAKE_BINARY_DIR}/input-order-checks")
        elseif(check STREQUAL "home_cache_renderer_checks")
          get_filename_component(spice_build_name "${CMAKE_BINARY_DIR}" NAME)
          set(check_arguments "${CMAKE_SOURCE_DIR}/.ai/evidence/native-checks/home-cache-${spice_build_name}")
        endif()
        add_test(NAME ${check}_native COMMAND ${CMAKE_COMMAND} -E env
          LIBGL_ALWAYS_SOFTWARE=1 MESA_GLES_VERSION_OVERRIDE=2.0
          ASAN_OPTIONS=detect_leaks=0:halt_on_error=1 UBSAN_OPTIONS=halt_on_error=1
          ${SPICE_XVFB_RUN} -a -s "-screen 0 1920x1080x24"
          $<TARGET_FILE:spice_${check}> ${check_arguments})
        set_tests_properties(${check}_native PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
          TIMEOUT 120 LABELS "native-renderer;address-ub-only;ui-leaks-disabled")
      endif()
    endforeach()
  endif()
  install(TARGETS spice-home-cpp RUNTIME DESTINATION .)
  install(DIRECTORY assets DESTINATION .)
  install(FILES appinfo.json icon.png LICENSE DESTINATION .)
endif()
